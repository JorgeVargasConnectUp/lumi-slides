Paso 3 · la idea general

Capas por dentro, slices por fuera

  • Capas (los círculos), de adentro hacia afuera: domainqué son las cosas applicationqué puede hacer la app uilo que ve la persona infrastructurede dónde vienen los datos
  • La regla de oro: las dependencias apuntan hacia adentro. domain no conoce a nadie; todos lo conocen a él.
  • Slices (las líneas punteadas). Cada feature es una rebanada que atraviesa todas las capas: tratamientos, clientas, citas.

Paso 6 · ui/pages

La página, con filtros por categoría

ui/pages/treatments-page.tsx
import { useState } from 'react'

import { Button } from '@/components/ui/button'

import type { TreatmentCategory } from '../../domain/treatment'
import { TreatmentCard } from '../components/treatment-card'
import { useTreatments } from '../hooks/use-treatments'

type CategoryFilter = TreatmentCategory | 'all'

const CATEGORIES: CategoryFilter[] = ['all', 'facial', 'body', 'hair-removal', 'hands-and-feet']

export function TreatmentsPage() {
  const { treatments, isLoading, error } = useTreatments()
  const [category, setCategory] = useState<CategoryFilter>('all')

  // Not state: it is calculated from the state on every render
  const visibleTreatments =
    category === 'all' ? treatments : treatments.filter((t) => t.category === category)

  if (isLoading) return <p className="p-6 text-muted-foreground">Loading treatments...</p>
  if (error) return <p className="p-6 text-destructive">{error}</p>

  return (
    <main className="mx-auto flex max-w-5xl flex-col gap-6 p-6">
      <h1 className="text-3xl font-semibold">Treatments</h1>
      <div className="flex flex-wrap gap-2">
        {CATEGORIES.map((option) => (
          <Button
            key={option}
            variant={category === option ? 'default' : 'outline'}
            onClick={() => setCategory(option)}
          >
            {option}
          </Button>
        ))}
      </div>
      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
        {visibleTreatments.map((treatment) => (
          <TreatmentCard key={treatment.id} treatment={treatment} />
        ))}
      </div>
    </main>
  )
}

// domain/treatment-repository.ts

  findById(id: string): Promise<Treatment | null>

infrastructure/local-storage-treatment-repository.ts
const STORAGE_KEY = 'lumi-treatments'

export function createLocalStorageTreatmentRepository(
  seed: Treatment[],
): TreatmentRepository {
  function read(): Treatment[] {
    const saved = localStorage.getItem(STORAGE_KEY)
    return saved ? (JSON.parse(saved) as Treatment[]) : seed
  }

  function write(treatments: Treatment[]) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(treatments))
  }

  return {
    async findAll() { return read() },
    async findById(id) { return read().find((t) => t.id === id) ?? null },
    async create(input) {
      const treatment = { ...input, id: crypto.randomUUID() }
      write([...read(), treatment])
      return treatment
    },
    async update(treatment) {
      write(read().map((t) => (t.id === treatment.id ? treatment : t)))
      return treatment
    },
  }
}

import { addMinutes, areIntervalsOverlapping } from 'date-fns'

import type { Appointment, AppointmentStatus } from './appointment'

export function calculateEndsAt(startsAt: Date, durationMinutes: number): Date {
  return addMinutes(startsAt, durationMinutes)
}

// Back-to-back is allowed: 9:00-10:00 and 10:00-11:00 do not overlap
export function hasScheduleConflict(
  candidate: { start: Date; end: Date },
  appointments: Appointment[],
): boolean {
  return appointments
    .filter((appointment) => appointment.status !== 'cancelled')
    .some((appointment) =>
      areIntervalsOverlapping(candidate, {
        start: new Date(appointment.startsAt),
        end: new Date(appointment.endsAt),
      }),
    )
}

const NEXT_STATUSES: Record<AppointmentStatus, AppointmentStatus[]> = {
  scheduled: ['confirmed', 'cancelled'],
  confirmed: ['completed', 'cancelled', 'no-show'],
  completed: [],
  cancelled: [],
  'no-show': [],
}

export function getNextStatuses(status: AppointmentStatus): AppointmentStatus[] {
  return NEXT_STATUSES[status]
}
